def parse_text_numbers(text: str):
    """テキストから数値トークンだけ抽出（正規表現ベースで堅牢）"""
    if not text: return []
    nums=[]
    for s in re.findall(r"[-+]?\d*\.?\d+(?:[eE][-+]?\d+)?", text):
        try: nums.append(float(s))
        except: pass
    return nums

def detect_and_normalize_scale(arr):
    """
    0〜1 or 0〜100% のヒューリスティック判定。
    - 25%以上が1超 or 95%点が1.2超 → %と判断し/100
    """
    a=np.asarray(arr,dtype=float)
    if a.size==0: return a,"unknown"
    over1_ratio = (a>1.0).mean()
    p95 = np.nanpercentile(a, 95) if np.isfinite(a).any() else 0
    if over1_ratio>=0.25 or p95>1.2:
        return a/100.0, "percent"
    return a, "proportion"

# ===== 統計コア（STDEV.S / STDEV.P 切り替え） ====================================
def _std_log(logs):
    """ログ値の標準偏差。ddofをSTD_MODEで切替。"""
    if logs.size<=1: return 0.0
    dd = 1 if STD_MODE == "sample" else 0
    return float(np.std(logs, ddof=dd))

def shapiro_on_logs(rates_pos):
    """log(率)のShapiro、平均・σ、ログ配列を返す。"""
    logs=np.log(rates_pos)
    W,p=stats.shapiro(logs)
    mu=float(np.mean(logs))
    sd=_std_log(logs)
    return W,p,mu,sd,logs

def lognorm_quantile(mu, sd, upper_tail_pct):
    """上位p%のしきい値（元スケール）"""
    q=1.0 - upper_tail_pct/100.0
    z=stats.norm.ppf(q)
    return float(math.exp(mu + sd*z))

def lognorm_tail_pct_for_threshold(mu, sd, thr):
    """thresholdが上位何%か（元分布基準）"""
    if sd<=0: return 0.0
    z=(math.log(thr)-mu)/sd
    return float(100.0*(1.0 - stats.norm.cdf(z)))

# ===== QQオーバレイ基準作成（Original座標でRemoved/Cleanedを重ねる） ============
def _split_for_overlay(logs_orig, removed_values):
    if logs_orig.size==0: return None
    osm, osr = stats.probplot(logs_orig, dist="norm", fit=False)
    osr=np.asarray(osr)
    logs_removed = np.log(np.asarray(removed_values, dtype=float)) if removed_values else np.array([])
    tol=1e-12; available=list(range(osr.size)); idx_removed=[]
    if logs_removed.size>0:
        targets=np.sort(logs_removed)[::-1]
        for v in targets:
            chosen=None
            for j in available:
                if abs(osr[j]-v)<=tol: chosen=j; break
            if chosen is None and available:
                diffs=[abs(osr[j]-v) for j in available]
                chosen=available[int(np.argmin(diffs))]
            if chosen is not None:
                idx_removed.append(chosen); available.remove(chosen)
    idx_removed=np.array(sorted(idx_removed)) if len(idx_removed) else np.array([],dtype=int)
    idx_clean=np.array(sorted(available)) if len(available) else np.array([],dtype=int)
    return {"osm":np.asarray(osm), "osr":osr, "idx_clean":idx_clean, "idx_removed":idx_removed}

def train_ml_from_csv(train_csv_path, texts=TEXTS["ja"]):
    """CSV(A=Name,B=Rate,C=IsOutlier)を学習してモデル辞書を返す"""
    df = read_csv_df(train_csv_path)
    cols=list(df.columns)
    if len(cols)<3:
        raise ValueError(texts["ERR_TRAIN_COLS"])
    df = df.rename(columns={cols[0]:'Name', cols[1]:'Rate', cols[2]:'IsOutlier'})
    y = pd.to_numeric(df['IsOutlier'], errors="coerce").fillna(0).astype(int).values
    X, feat_cols = _ml_features_from_df(df[['Name','Rate']].copy())

    clf = GradientBoostingClassifier(random_state=RANDOM_STATE)
    # ざっくりCV表示（失敗しても無視）
    try:
        skf=StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)
        aucs=[]
        for tr,te in skf.split(X,y):
            clf.fit(X[tr], y[tr]); p=clf.predict_proba(X[te])[:,1]
            aucs.append(roc_auc_score(y[te], p))
        print(f"[ML] CV AUC: {np.mean(aucs):.3f} ± {np.std(aucs):.3f}")
    except Exception:
        pass
    clf.fit(X,y)
    artifact = {
        "model": clf,
        "feature_columns": feat_cols,
        "created_at": datetime.datetime.now().isoformat(timespec="seconds"),
        "version": APP_VERSION,
        "algo": "GradientBoostingClassifier"
    }
    return artifact

# ===== 総合計算（外れ値モード込み） =============================================
def compute_all_with_mode(rates_raw, tail_pct, alpha,
                          include_zero_in_overall=False,
                          mode="none",
                          ml_artifact=None, ml_threshold=0.5,
                          max_remove_frac=DEFAULT_MAX_REMOVE_FRAC,
                          return_ml_flags=False,
                          texts=TEXTS["ja"]):
    out={"messages":[]}
    # スケール正規化
    arr, scale = detect_and_normalize_scale(rates_raw)
    arr = np.asarray(arr, dtype=float); arr = arr[np.isfinite(arr)]
    arr = arr[(arr>=0.0) & (arr<=1.0)]

    # 最低限チェック
    ok_basic, reasons, N_total, N_pos = validate_counts(arr, texts)
    out.update(dict(scale=scale, n_total=N_total, n_pos=N_pos))
    out["messages"].extend(reasons); out["basic_ok"]=ok_basic

    pos = arr[arr>0]

    # 0%含む全体補正（Excel準拠）
    eff_pct = float(tail_pct)*(N_total/N_pos) if (include_zero_in_overall and N_pos>0) else float(tail_pct)
    out["effective_tail_pct"]=eff_pct; out["include_zero_in_overall"]=include_zero_in_overall
    out["a_over_100"] = (eff_pct>=100.0)

    # Original（除外なし）
    orig = {"W":np.nan,"p":np.nan,"mu":np.nan,"sd":np.nan,"thr":np.nan,"logs":np.array([])}
    if pos.size>=3:
        try:
            W,p,mu,sd,logs = shapiro_on_logs(pos)
            if eff_pct<=0: thr=float("nan")
            elif eff_pct>=100.0: thr=0.0
            else: thr = math.exp(mu) if sd<=0 else lognorm_quantile(mu, sd, eff_pct)
            orig.update(dict(W=W,p=p,mu=mu,sd=sd,thr=thr,logs=logs))
        except Exception as e:
            out["messages"].append(texts["MSG_SHAPIRO_ERR_ORIG"].format(err=e))
    else:
        out["messages"].append(texts["MSG_LOG_DATA_SHORT"])

    # 外れ値除外（逐次 or ML）
    removed = []
    if pos.size>=3:
        if mode == "none":
            stop_reason = texts["MSG_MODE_NONE"]
            cleaned_arr = pos.copy()
            ml_flag_indices = set()

        elif mode == "sequential":
            working = pos.copy()
            max_remove = int(math.floor(pos.size * max_remove_frac))
            step=0
            while True:
                try:
                    Wc,pc,muc,sdc,logc = shapiro_on_logs(working)
                except Exception as e:
                    out["messages"].append(texts["MSG_SHAPIRO_ERR_SEQ"].format(err=e))
                    break
                if pc>=alpha:
                    stop_reason=texts["MSG_OK"].format(p=pc, alpha=alpha)
                    cleaned_arr=working.copy(); break
                if step>=max_remove or working.size<=3:
                    stop_reason=texts["MSG_MODE_SEQ_LIMIT"]; cleaned_arr=working.copy(); break
                idx_max=int(np.argmax(working))
                removed.append(float(working[idx_max]))
                working = np.delete(working, idx_max); step+=1
            ml_flag_indices = set()

        elif mode == "ml":
            if (ml_artifact is None):
                stop_reason=texts["MSG_MODE_ML_NONE"]
                cleaned_arr = pos.copy(); ml_flag_indices=set()
            else:
                df_tmp = pd.DataFrame({"Name":["SINGLE"]*len(pos), "Rate":pos})
                X, _ = _ml_features_from_df(df_tmp[['Name','Rate']].copy())
                model = ml_artifact.get("model", ml_artifact)
                proba = model.predict_proba(X)[:,1]
                flag = (proba >= float(ml_threshold))
                ml_flag_indices = set(np.where(flag)[0].tolist())
                idx_flag = np.where(flag)[0]
                max_remove = int(math.floor(pos.size * max_remove_frac))
                if idx_flag.size > max_remove:
                    order = np.argsort(proba[idx_flag])[::-1]
                    idx_flag = idx_flag[order[:max_remove]]
                mask = np.ones(pos.size, dtype=bool); mask[idx_flag] = False
                cleaned_arr = pos[mask]
                removed = pos[~mask].tolist()
                stop_reason = texts["MSG_MODE_ML_USED"].format(thr=ml_threshold)
    else:
        cleaned_arr = pos.copy()
        stop_reason = texts["MSG_LOG_DATA_SHORT"]
        ml_flag_indices = set()

    # Cleaned 統計
    cleaned = {"W":np.nan,"p":np.nan,"mu":np.nan,"sd":np.nan,"thr":np.nan,"logs":np.array([]),
               "removed_values":removed, "stop_reason":stop_reason}
    if cleaned_arr.size>=3:
        try:
            Wc,pc,muc,sdc,logc = shapiro_on_logs(cleaned_arr)
            thc = float("nan") if eff_pct<=0 else (0.0 if eff_pct>=100.0 else (math.exp(muc) if sdc<=0 else lognorm_quantile(muc,sdc,eff_pct)))
            cleaned.update(dict(W=Wc,p=pc,mu=muc,sd=sdc,thr=thc,logs=logc))
        except Exception as e:
            out["messages"].append(texts["MSG_SHAPIRO_ERR_CLEAN"].format(err=e))

    out["orig"]=orig; out["cleaned"]=cleaned
    overall_ok = ok_basic and np.isfinite(cleaned.get("p",np.nan)) and (cleaned["p"]>=alpha)
    out["overall_ok"]=overall_ok
    if overall_ok and np.isfinite(cleaned.get("p",np.nan)):
        out["messages"].append(texts["MSG_OK"].format(p=cleaned['p'], alpha=alpha))
    else:
        if not ok_basic:
            pass
        elif np.isfinite(cleaned.get("p",np.nan)):
            out["messages"].append(texts["MSG_NG"].format(p=cleaned['p'], alpha=alpha))
        else:
            out["messages"].append(texts["MSG_P_NA"])

    # 元しきい値で外れが弾けるか→最小しきい値/対応%の算出
    outlier_check={"orig_thr":orig["thr"], "missed_outliers":[], "required_thr":None, "required_tail_pct":None}
    if len(removed)>0 and np.isfinite(orig["thr"]):
        removed_vals = sorted(removed, reverse=True)
        missed = [v for v in removed_vals if not (v>=orig["thr"])]
        outlier_check["missed_outliers"]=missed
        if len(missed)>0:
            req_thr = min(missed); outlier_check["required_thr"]=req_thr
            req_tail = lognorm_tail_pct_for_threshold(orig["mu"], orig["sd"], req_thr) if (np.isfinite(orig["sd"]) and orig["sd"]>0) else 0.0
            outlier_check["required_tail_pct"]=req_tail
    out["outlier_check"]=outlier_check

    if return_ml_flags:
        out["ml_flags_indices"] = ml_flag_indices
    return out

# ===== 一括処理（DataFrame→集計） ==============================================
def batch_from_dataframe(df, key_col, rate_col, tail_pct, alpha,
                         include_zero_in_overall, mode, ml_artifact, ml_threshold,
                         max_remove_frac, std_mode_label, texts):
    rows = []
    if key_col not in df.columns or rate_col not in df.columns:
        raise ValueError(texts["ERR_SELECT_COLUMNS"])
    tmp = pd.DataFrame({"key": df[key_col].astype(str),
                        "rate": pd.to_numeric(df[rate_col], errors="coerce")}).dropna(subset=["rate"])
    if tmp.empty:
        return pd.DataFrame()
    for key, g in tmp.groupby("key"):
        rates = g["rate"].tolist()
        try:
            res = compute_all_with_mode(
                rates, tail_pct, alpha,
                include_zero_in_overall=include_zero_in_overall,
                mode=mode,
                ml_artifact=ml_artifact,
                ml_threshold=ml_threshold,
                max_remove_frac=max_remove_frac,
                return_ml_flags=False,
                texts=texts
            )
            cleaned = res["cleaned"]; orig = res["orig"]; oc = res["outlier_check"]
            rows.append({
                "Name": key,
                "N_total": len(rates),
                "N_pos": int(np.sum(np.asarray(rates) > 0)),
                "Scale": res["scale"],
                "StdMode": std_mode_label,
                "OutlierMode": mode,
                "MaxRemoveFrac": max_remove_frac,
                "IncludeZeroOverall": include_zero_in_overall,
                "TailPct_Input": tail_pct,
                "TailPct_Effective": res.get("effective_tail_pct", tail_pct),
                "ShapiroW_Cleaned": cleaned.get("W", np.nan),
                "pvalue_Cleaned": cleaned.get("p", np.nan),
                "W_Original": orig.get("W", np.nan),
                "p_Original": orig.get("p", np.nan),
                "Threshold_Cleaned": cleaned.get("thr", np.nan),
                "Threshold_Original": oc.get("orig_thr", np.nan),
                "MinThreshold_RejectOutliers": oc.get("required_thr", np.nan),
                "MinThreshold_TailPct_onOriginal": oc.get("required_tail_pct", np.nan),
                "RemovedCount": len(cleaned.get("removed_values", [])),
                "RemovedTop10": ", ".join([f"{v:.8f}" for v in cleaned.get("removed_values", [])[:10]]),
                "StopReason": cleaned.get("stop_reason", ""),
                "OverallOK": res.get("overall_ok", False),
                "Messages": " / ".join(res.get("messages", [])),
                "RunAt": datetime.datetime.now().isoformat(timespec="seconds"),
                "AppVersion": APP_VERSION,
                "RandomSeed": RANDOM_STATE
            })
        except Exception as e:
            rows.append({"Name": key, "Error": str(e),
                         "RunAt": datetime.datetime.now().isoformat(timespec="seconds"),
                         "AppVersion": APP_VERSION,
                         "RandomSeed": RANDOM_STATE})
    out_df = pd.DataFrame(rows)
    preferred = [
        "Name","N_total","N_pos","Scale","StdMode","OutlierMode","MaxRemoveFrac",
        "IncludeZeroOverall","TailPct_Input","TailPct_Effective",
        "ShapiroW_Cleaned","pvalue_Cleaned","W_Original","p_Original",
        "Threshold_Cleaned","Threshold_Original",
        "MinThreshold_RejectOutliers","MinThreshold_TailPct_onOriginal",
        "RemovedCount","StopReason","RemovedTop10","OverallOK","Messages",
        "RunAt","AppVersion","RandomSeed","Error"
    ]
    for c in preferred:
        if c not in out_df.columns: out_df[c] = np.nan
    out_df = out_df[preferred]
    return out_df

def detect_and_normalize_scale(arr):
    """
    0〜1 or 0〜100% のヒューリスティック判定。
    - 25%以上が1超 or 95%点が1.2超 → %と判断し/100
    """
    a=np.asarray(arr,dtype=float)
    if a.size==0: return a,"unknown"
    over1_ratio = (a>1.0).mean()
    p95 = np.nanpercentile(a, 95) if np.isfinite(a).any() else 0
    if over1_ratio>=0.25 or p95>1.2:
        return a/100.0, "percent"
    return a, "proportion"

# ===== 統計コア（STDEV.S / STDEV.P 切り替え） ====================================
def _std_log(logs):
    """ログ値の標準偏差。ddofをSTD_MODEで切替。"""
    if logs.size<=1: return 0.0
    dd = 1 if STD_MODE == "sample" else 0
    return float(np.std(logs, ddof=dd))

def shapiro_on_logs(rates_pos):
    """log(率)のShapiro、平均・σ、ログ配列を返す。"""
    logs=np.log(rates_pos)
    W,p=stats.shapiro(logs)
    mu=float(np.mean(logs))
    sd=_std_log(logs)
    return W,p,mu,sd,logs

def lognorm_quantile(mu, sd, upper_tail_pct):
    """上位p%のしきい値（元スケール）"""
    q=1.0 - upper_tail_pct/100.0
    z=stats.norm.ppf(q)
    return float(math.exp(mu + sd*z))

def lognorm_tail_pct_for_threshold(mu, sd, thr):
    """thresholdが上位何%か（元分布基準）"""
    if sd<=0: return 0.0
    z=(math.log(thr)-mu)/sd
    return float(100.0*(1.0 - stats.norm.cdf(z)))

# ===== QQオーバレイ基準作成 =====================================================
def _split_for_overlay(logs_orig, removed_values):
    if logs_orig.size==0: return None
    osm, osr = stats.probplot(logs_orig, dist="norm", fit=False)
    osr=np.asarray(osr)
    logs_removed = np.log(np.asarray(removed_values, dtype=float)) if removed_values else np.array([])
    tol=1e-12; available=list(range(osr.size)); idx_removed=[]
    if logs_removed.size>0:
        targets=np.sort(logs_removed)[::-1]
        for v in targets:
            chosen=None
            for j in available:
                if abs(osr[j]-v)<=tol: chosen=j; break
            if chosen is None and available:
                diffs=[abs(osr[j]-v) for j in available]
                chosen=available[int(np.argmin(diffs))]
            if chosen is not None:
                idx_removed.append(chosen); available.remove(chosen)
    idx_removed=np.array(sorted(idx_removed)) if len(idx_removed) else np.array([],dtype=int)
    idx_clean=np.array(sorted(available)) if len(available) else np.array([],dtype=int)
    return {"osm":np.asarray(osm), "osr":osr, "idx_clean":idx_clean, "idx_removed":idx_removed}

# ===== 総合計算（外れ値モード込み） =============================================
def compute_all_with_mode(rates_raw, tail_pct, alpha,
                          include_zero_in_overall=False,
                          mode="none",
                          ml_artifact=None, ml_threshold=0.5,
                          max_remove_frac=DEFAULT_MAX_REMOVE_FRAC,
                          return_ml_flags=False,
                          texts=TEXTS["ja"]):
    out={"messages":[]}
    # スケール正規化
    arr, scale = detect_and_normalize_scale(rates_raw)
    arr = np.asarray(arr, dtype=float); arr = arr[np.isfinite(arr)]
    arr = arr[(arr>=0.0) & (arr<=1.0)]

    # 最低限チェック
    ok_basic, reasons, N_total, N_pos = validate_counts(arr, texts)
    out.update(dict(scale=scale, n_total=N_total, n_pos=N_pos))
    out["messages"].extend(reasons); out["basic_ok"]=ok_basic

    pos = arr[arr>0]

    # 0%含む全体補正（Excel準拠）
    eff_pct = float(tail_pct)*(N_total/N_pos) if (include_zero_in_overall and N_pos>0) else float(tail_pct)
    out["effective_tail_pct"]=eff_pct; out["include_zero_in_overall"]=include_zero_in_overall
    out["a_over_100"] = (eff_pct>=100.0)

    # Original（除外なし）
    orig = {"W":np.nan,"p":np.nan,"mu":np.nan,"sd":np.nan,"thr":np.nan,"logs":np.array([])}
    if pos.size>=3:
        try:
            W,p,mu,sd,logs = shapiro_on_logs(pos)
            if eff_pct<=0: thr=float("nan")
            elif eff_pct>=100.0: thr=0.0
            else: thr = math.exp(mu) if sd<=0 else lognorm_quantile(mu, sd, eff_pct)
            orig.update(dict(W=W,p=p,mu=mu,sd=sd,thr=thr,logs=logs))
        except Exception as e:
            out["messages"].append(texts["MSG_SHAPIRO_ERR_ORIG"].format(err=e))
    else:
        out["messages"].append(texts["MSG_LOG_DATA_SHORT"])

    # 外れ値除外（逐次 or ML）
    removed = []
    if pos.size>=3:
        if mode == "none":
            stop_reason = texts["MSG_MODE_NONE"]
            cleaned_arr = pos.copy()
            ml_flag_indices = set()

        elif mode == "sequential":
            working = pos.copy()
            max_remove = int(math.floor(pos.size * max_remove_frac))
            step=0
            while True:
                try:
                    Wc,pc,muc,sdc,logc = shapiro_on_logs(working)
                except Exception as e:
                    out["messages"].append(texts["MSG_SHAPIRO_ERR_SEQ"].format(err=e))
                    break
                if pc>=alpha:
                    stop_reason=texts["MSG_OK"].format(p=pc, alpha=alpha)
                    cleaned_arr=working.copy(); break
                if step>=max_remove or working.size<=3:
                    stop_reason=texts["MSG_MODE_SEQ_LIMIT"]; cleaned_arr=working.copy(); break
                idx_max=int(np.argmax(working))
                removed.append(float(working[idx_max]))
                working = np.delete(working, idx_max); step+=1
            ml_flag_indices = set()

        elif mode == "ml":
            if (ml_artifact is None):
                stop_reason=texts["MSG_MODE_ML_NONE"]
                cleaned_arr = pos.copy(); ml_flag_indices=set()
            else:
                df_tmp = pd.DataFrame({"Name":["SINGLE"]*len(pos), "Rate":pos})
                X, _ = _ml_features_from_df(df_tmp[['Name','Rate']].copy())
                model = ml_artifact.get("model", ml_artifact)
                proba = model.predict_proba(X)[:,1]
                flag = (proba >= float(ml_threshold))
                ml_flag_indices = set(np.where(flag)[0].tolist())
                idx_flag = np.where(flag)[0]
                max_remove = int(math.floor(pos.size * max_remove_frac))
                if idx_flag.size > max_remove:
                    order = np.argsort(proba[idx_flag])[::-1]
                    idx_flag = idx_flag[order[:max_remove]]
                mask = np.ones(pos.size, dtype=bool); mask[idx_flag] = False
                cleaned_arr = pos[mask]
                removed = pos[~mask].tolist()
                stop_reason = texts["MSG_MODE_ML_USED"].format(thr=ml_threshold)
    else:
        cleaned_arr = pos.copy()
        stop_reason = texts["MSG_LOG_DATA_SHORT"]
        ml_flag_indices = set()

    # Cleaned 統計
    cleaned = {"W":np.nan,"p":np.nan,"mu":np.nan,"sd":np.nan,"thr":np.nan,"logs":np.array([]),
               "removed_values":removed, "stop_reason":stop_reason}
    if cleaned_arr.size>=3:
        try:
            Wc,pc,muc,sdc,logc = shapiro_on_logs(cleaned_arr)
            thc = float("nan") if eff_pct<=0 else (0.0 if eff_pct>=100.0 else (math.exp(muc) if sdc<=0 else lognorm_quantile(muc,sdc,eff_pct)))
            cleaned.update(dict(W=Wc,p=pc,mu=muc,sd=sdc,thr=thc,logs=logc))
        except Exception as e:
            out["messages"].append(texts["MSG_SHAPIRO_ERR_CLEAN"].format(err=e))

    out["orig"]=orig; out["cleaned"]=cleaned
    overall_ok = ok_basic and np.isfinite(cleaned.get("p",np.nan)) and (cleaned["p"]>=alpha)
    out["overall_ok"]=overall_ok
    if overall_ok and np.isfinite(cleaned.get("p",np.nan)):
        out["messages"].append(texts["MSG_OK"].format(p=cleaned['p'], alpha=alpha))
    else:
        if not ok_basic:
            pass
        elif np.isfinite(cleaned.get("p",np.nan)):
            out["messages"].append(texts["MSG_NG"].format(p=cleaned['p'], alpha=alpha))
        else:
            out["messages"].append(texts["MSG_P_NA"])

    # 元しきい値で外れが弾けるか→最小しきい値/対応%の算出
    outlier_check={"orig_thr":orig["thr"], "missed_outliers":[], "required_thr":None, "required_tail_pct":None}
    if len(removed)>0 and np.isfinite(orig["thr"]):
        removed_vals = sorted(removed, reverse=True)
        missed = [v for v in removed_vals if not (v>=orig["thr"])]
        outlier_check["missed_outliers"]=missed
        if len(missed)>0:
            req_thr = min(missed); outlier_check["required_thr"]=req_thr
            req_tail = lognorm_tail_pct_for_threshold(orig["mu"], orig["sd"], req_thr) if (np.isfinite(orig["sd"]) and orig["sd"]>0) else 0.0
            outlier_check["required_tail_pct"]=req_tail
    out["outlier_check"]=outlier_check

    if return_ml_flags:
        out["ml_flags_indices"] = ml_flag_indices
    return out

# ===== 一括処理（DataFrame→集計） ==============================================
def batch_from_dataframe(df, key_col, rate_col, tail_pct, alpha,
                         include_zero_in_overall, mode, ml_artifact, ml_threshold,
                         max_remove_frac, std_mode_label, texts):
    rows = []
    if key_col not in df.columns or rate_col not in df.columns:
        raise ValueError(texts["ERR_SELECT_COLUMNS"])
    tmp = pd.DataFrame({"key": df[key_col].astype(str),
                        "rate": pd.to_numeric(df[rate_col], errors="coerce")}).dropna(subset=["rate"])
    if tmp.empty:
        return pd.DataFrame()
    for key, g in tmp.groupby("key"):
        rates = g["rate"].tolist()
        try:
            res = compute_all_with_mode(
                rates, tail_pct, alpha,
                include_zero_in_overall=include_zero_in_overall,
                mode=mode,
                ml_artifact=ml_artifact,
                ml_threshold=ml_threshold,
                max_remove_frac=max_remove_frac,
                return_ml_flags=False,
                texts=texts
            )
            cleaned = res["cleaned"]; orig = res["orig"]; oc = res["outlier_check"]
            rows.append({
                "Name": key,
                "N_total": len(rates),
                "N_pos": int(np.sum(np.asarray(rates) > 0)),
                "Scale": res["scale"],
                "StdMode": std_mode_label,
                "OutlierMode": mode,
                "MaxRemoveFrac": max_remove_frac,
                "IncludeZeroOverall": include_zero_in_overall,
                "TailPct_Input": tail_pct,
                "TailPct_Effective": res.get("effective_tail_pct", tail_pct),
                "ShapiroW_Cleaned": cleaned.get("W", np.nan),
                "pvalue_Cleaned": cleaned.get("p", np.nan),
                "W_Original": orig.get("W", np.nan),
                "p_Original": orig.get("p", np.nan),
                "Threshold_Cleaned": cleaned.get("thr", np.nan),
                "Threshold_Original": oc.get("orig_thr", np.nan),
                "MinThreshold_RejectOutliers": oc.get("required_thr", np.nan),
                "MinThreshold_TailPct_onOriginal": oc.get("required_tail_pct", np.nan),
                "RemovedCount": len(cleaned.get("removed_values", [])),
                "RemovedTop10": ", ".join([f"{v:.8f}" for v in cleaned.get("removed_values", [])[:10]]),
                "StopReason": cleaned.get("stop_reason", ""),
                "OverallOK": res.get("overall_ok", False),
                "Messages": " / ".join(res.get("messages", [])),
                "RunAt": datetime.datetime.now().isoformat(timespec="seconds"),
                "AppVersion": APP_VERSION,
                "RandomSeed": RANDOM_STATE
            })
        except Exception as e:
            rows.append({"Name": key, "Error": str(e),
                         "RunAt": datetime.datetime.now().isoformat(timespec="seconds"),
                         "AppVersion": APP_VERSION,
                         "RandomSeed": RANDOM_STATE})
    out_df = pd.DataFrame(rows)
    preferred = [
        "Name","N_total","N_pos","Scale","StdMode","OutlierMode","MaxRemoveFrac",
        "IncludeZeroOverall","TailPct_Input","TailPct_Effective",
        "ShapiroW_Cleaned","pvalue_Cleaned","W_Original","p_Original",
        "Threshold_Cleaned","Threshold_Original",
        "MinThreshold_RejectOutliers","MinThreshold_TailPct_onOriginal",
        "RemovedCount","StopReason","RemovedTop10","OverallOK","Messages",
        "RunAt","AppVersion","RandomSeed","Error"
    ]
    for c in preferred:
        if c not in out_df.columns: out_df[c] = np.nan
    out_df = out_df[preferred]
    return out_df

def pick_name_modal(parent_window, names, texts):
    """
    親ウィンドウ中央にモーダルで出す品名選択ダイアログ。
    - 検索ボックスで絞り込み
    - OKで選択確定 / キャンセルでNone

    if not names:
        return None

    # 親の位置/サイズから中央座標を計算し、画面内にクランプ
    root = parent_window.TKroot
    root.update_idletasks()
    px = root.winfo_rootx()
    py = root.winfo_rooty()
    pw = root.winfo_width()
    ph = root.winfo_height()
    sw = root.winfo_screenwidth()
    sh = root.winfo_screenheight()

    win_w, win_h = 420, 420
    cx = px + max(0, pw//2 - win_w//2)
    cy = py + max(0, ph//2 - win_h//2)
    cx = max(0, min(cx, sw - win_w))
    cy = max(0, min(cy, sh - win_h))

    # 初期値

        [sg.Text(texts["BVIEW_PICK_PROMPT"])],
        [sg.Text(texts["BVIEW_SEARCH"]), sg.Input(key="-PICK-FLT-", enable_events=True, size=(28,1))],
        [sg.Listbox(values=filtered, key="-PICK-LB-", size=(40,15), enable_events=True, select_mode=sg.LISTBOX_SELECT_MODE_SINGLE)],
        [sg.Push(), sg.Button(texts["BVIEW_OK"], key="-PICK-OK-", bind_return_key=True),
         sg.Button(texts["BVIEW_CANCEL"], key="-PICK-CANCEL-")]
    ]

    win = sg.Window("Picker", layout, modal=True, keep_on_top=True,
                    location=(cx, cy), finalize=True)

    selected = None
    while True:
        ev, vals = win.read()
        if ev in (sg.WIN_CLOSED, "-PICK-CANCEL-"):
            selected = None
            break
        elif ev == "-PICK-FLT-":
            kw = (vals["-PICK-FLT-"] or "").strip().lower()
            if kw:
                filtered = [n for n in names if kw in n.lower()]
            else:
                filtered = list(names)
            win["-PICK-LB-"].update(filtered)
            if not filtered:
                win["-PICK-LB-"].update([f"({texts['BVIEW_NO_MATCH']})"])

            # クリックで選択ハイライトのみ。確定はOKで。
            pass
        elif ev == "-PICK-OK-":
            vals_now = win["-PICK-LB-"].get_list_values()
            sel = win["-PICK-LB-"].get()
            if sel and sel[0] in vals_now and not sel[0].startswith("("):
                selected = sel[0]
                break
    win.close()
    return selected

# ---- PyInstaller 同梱フォント（最終手段） ----
MEI = getattr(sys, "_MEIPASS", None)
def _bundled_font_path(filename):
    if MEI:
        return os.path.join(MEI, "fonts", filename)
    return os.path.join(os.path.dirname(__file__), "fonts", filename)

# ja: NotoSansCJKjp, zh: NotoSansCJKsc を最後に差し込む
LANG_BUNDLED_FALLBACK = {
    "ja": _bundled_font_path("NotoSansCJKjp-Regular.otf"),
    "zh": _bundled_font_path("NotoSansCJKsc-Regular.otf"),
}

def select_font_for_lang(lang: str) -> str:
    """
    言語(lang)に応じて候補フォントを探索して設定。
    見つかったフォント名を返す（見つからなければ 'sans-serif'）。
    """
    candidates = list(LANG_FONT_CANDIDATES.get(lang, []))
    bundled = LANG_BUNDLED_FALLBACK.get(lang)
    if bundled:
        # 同梱フォントは最後の“最終手段”として追加
        candidates.append(bundled)

    for p in candidates:
        if p and os.path.exists(p):
            try:
                font_manager.fontManager.addfont(p)
                name = font_manager.FontProperties(fname=p).get_name()
                matplotlib.rcParams["font.family"] = name
                print(f"[Font] {lang} -> {name} ({p})")
                return name
            except Exception:
                continue

    print(f"[Font] {lang} 用フォント未発見。デフォルト使用。")
    return "sans-serif"

                window[key].update(TEXTS[lang][key])
